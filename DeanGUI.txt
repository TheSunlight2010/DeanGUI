-- Animation Ownership Bypass Attempt
local function tryLoadAnimation(animId)
    local anim = Instance.new("Animation")
    anim.AnimationId = animId
    
    -- Attempt 1: Spoof creator if possible
    local success = pcall(function()
        if sethiddenproperty then
            -- Try to spoof the internal creator ID to match the game
            sethiddenproperty(anim, "Creator", game.CreatorId)
        end
    end)
    
    if not success then
        -- Fallback: try alternative property names
        pcall(function()
            if sethiddenproperty then
                sethiddenproperty(anim, "CreatorId", game.CreatorId)
            end
        end)
    end
    
    -- Attempt to load
    local track = animator:LoadAnimation(anim)
    
    -- If load succeeded, destroy the template (track keeps reference)
    if track then
        anim:Destroy()
        return track
    else
        anim:Destroy()
        return nil
    end
end

-- Animation State
local currentTrack = nil
local currentAnimName = nil
local isBlowPlaying = false

-- Function to stop current animation
local function stopCurrentAnimation()
    if currentTrack then
        currentTrack:Stop(0.2)
        currentTrack = nil
    end
    currentAnimName = nil
    isBlowPlaying = false
    StatusLabel.Text = "Status: Idle"
    StatusLabel.TextColor3 = Color3.fromRGB(135, 206, 235)
    
    -- Re-enable movement if it was disabled
    if humanoid then
        humanoid.WalkSpeed = 16
        humanoid.JumpPower = 50
        humanoid.AutoRotate = true
    end
end

-- Function to play animation (with ownership bypass)
local function playAnimation(animName, animData)
    -- Stop current animation first (smooth transition)
    if currentTrack then
        currentTrack:Stop(0.15)
        task.wait(0.15)
    end
    
    local track = nil
    
    -- Try bypass method first for non-Roblox anims
    -- (You can detect "official" vs custom by checking if it fails normally)
    local normalAnim = Instance.new("Animation")
    normalAnim.AnimationId = animData.id
    local normalSuccess, normalResult = pcall(function()
        return animator:LoadAnimation(normalAnim)
    end)
    normalAnim:Destroy()
    
    if normalSuccess and normalResult then
        -- Official anim or already-allowed, use normal load
        track = normalResult
        print("[ICEQUEEN] Loaded normally: " .. animName)
    else
        -- Failed normal load, try bypass
        print("[ICEQUEEN] Normal load failed, attempting bypass for: " .. animName)
        local bypassSuccess, bypassResult = pcall(function()
            return tryLoadAnimation(animData.id)
        end)
        
        if bypassSuccess and bypassResult then
            track = bypassResult
            print("[ICEQUEEN] Bypass succeeded: " .. animName)
        else
            warn("[ICEQUEEN] Bypass failed for: " .. animName)
            StatusLabel.Text = "Status: Failed to load \"" .. animName .. "\""
            StatusLabel.TextColor3 = Color3.fromRGB(255, 100, 100)
            return
        end
    end
    
    -- Configure track
    track.Priority = animData.priority or Enum.AnimationPriority.Action
    track.Looped = animData.loop or false
    track:Play(0.2)
    
    currentTrack = track
    currentAnimName = animName
    
    -- Update status
    StatusLabel.Text = "Status: Playing \"" .. animName .. "\""
    StatusLabel.TextColor3 = Color3.fromRGB(100, 255, 100)
    
    -- Handle Blow special case (cannot move, pauses at end)
    if animName == "Blow" then
        isBlowPlaying = true
        humanoid.WalkSpeed = 0
        humanoid.JumpPower = 0
        humanoid.AutoRotate = false
        
        -- Wait for animation to finish
        track.Stopped:Connect(function()
            if currentAnimName == "Blow" then
                -- Pause at last frame
                track.TimePosition = track.Length - 0.05
                StatusLabel.Text = "Status: Blow (Paused at end)"
                StatusLabel.TextColor3 = Color3.fromRGB(255, 200, 100)
            end
        end)
    else
        isBlowPlaying = false
        humanoid.WalkSpeed = 16
        humanoid.JumpPower = 50
        humanoid.AutoRotate = true
    end
end

-- Drag functionality (same as Infinite Yield)
local function dragGUI(gui)
    task.spawn(function()
        local dragging
        local dragInput
        local dragStart = Vector3.new(0,0,0)
        local startPos
        local function update(input)
            local delta = input.Position - dragStart
            local Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
            TweenService:Create(gui, TweenInfo.new(.20), {Position = Position}):Play()
        end
        gui.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                dragging = true
                dragStart = input.Position
                startPos = gui.Position

                input.Changed:Connect(function()
                    if input.UserInputState == Enum.UserInputState.End then
                        dragging = false
                    end
                end)
            end
        end)
        gui.InputChanged:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
                dragInput = input
            end
        end)
        UserInputService.InputChanged:Connect(function(input)
            if input == dragInput and dragging then
                update(input)
            end
        end)
    end)
end
