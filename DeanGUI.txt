local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")

local player = Players.LocalPlayer

local DASH_ANIMATION_ID = "rbxassetid://104716596249497"
local MIN_DISTANCE = 10
local MAX_DISTANCE = 45
local DASH_SPEED = 100
local WALL_BOUNCE_ANGLE = 0.35
local COOLDOWN = 0.5

local dashDistance = 25
local canDash = true
local dashing = false

local character
local humanoid
local root
local animator
local animationTrack

-- Animation
local animation = Instance.new("Animation")
animation.AnimationId = DASH_ANIMATION_ID

local function setupCharacter(char)
	character = char
	humanoid = char:WaitForChild("Humanoid")
	root = char:WaitForChild("HumanoidRootPart")
	animator = humanoid:WaitForChild("Animator")

	-- Stop old track if necessary
	if animationTrack then
		animationTrack:Stop(0)
		animationTrack = nil
	end

	-- Load animation through the character's Animator
	animationTrack = animator:LoadAnimation(animation)
	animationTrack.Priority = Enum.AnimationPriority.Action
	animationTrack.Looped = false
end

setupCharacter(player.Character or player.CharacterAdded:Wait())

player.CharacterAdded:Connect(function(char)
	dashing = false
	canDash = true
	setupCharacter(char)
end)

local function playDashAnimation()
	if not animationTrack then
		return
	end

	if animationTrack.IsPlaying then
		animationTrack:Stop(0.05)
	end

	animationTrack:Play(0.05)
end

local function stopDashAnimation()
	if animationTrack and animationTrack.IsPlaying then
		animationTrack:Stop(0.1)
	end
end

local draggingSlider = false

local function updateSlider(mouseX)
	local absoluteX = sliderBackground.AbsolutePosition.X
	local width = sliderBackground.AbsoluteSize.X

	local percentage = math.clamp(
		(mouseX - absoluteX) / width,
		0,
		1
	)

	dashDistance = math.floor(
		MIN_DISTANCE
			+ (MAX_DISTANCE - MIN_DISTANCE) * percentage
			+ 0.5
	)

	distanceLabel.Text = "Dash Distance: " .. dashDistance

	sliderButton.Position = UDim2.new(
		percentage,
		0,
		0.5,
		0
	)
end

sliderButton.MouseButton1Down:Connect(function()
	draggingSlider = true
end)

sliderBackground.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		updateSlider(input.Position.X)
		draggingSlider = true
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if draggingSlider
		and input.UserInputType == Enum.UserInputType.MouseMovement then

		updateSlider(input.Position.X)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 then
		draggingSlider = false
	end
end)

-- GUI dragging
local draggingGui = false
local dragStart
local startPosition

local function updateGuiDrag(input)
	local delta = input.Position - dragStart

	main.Position = UDim2.new(
		startPosition.X.Scale,
		startPosition.X.Offset + delta.X,
		startPosition.Y.Scale,
		startPosition.Y.Offset + delta.Y
	)
end

title.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		draggingGui = true
		dragStart = input.Position
		startPosition = main.Position
	end
end)

UserInputService.InputChanged:Connect(function(input)
	if draggingGui
		and (
			input.UserInputType == Enum.UserInputType.MouseMovement
			or input.UserInputType == Enum.UserInputType.Touch
		) then

		updateGuiDrag(input)
	end
end)

UserInputService.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1
		or input.UserInputType == Enum.UserInputType.Touch then

		draggingGui = false
	end
end)

-- Raycast
local rayParams = RaycastParams.new()
rayParams.FilterType = Enum.RaycastFilterType.Exclude

local function getBounceDirection(direction, normal)
	local reflected =
		direction - 2 * direction:Dot(normal) * normal

	if math.abs(direction:Dot(normal)) < WALL_BOUNCE_ANGLE then
		return reflected.Unit
	end

	return nil
end

-- Dash
local function dash()
	if not canDash or dashing then
		return
	end

	if not character or not humanoid or not root then
		return
	end

	if humanoid.Health <= 0 then
		return
	end

	canDash = false
	dashing = true

	-- This is played through the character's Animator,
	-- allowing Roblox's normal character animation
	-- replication to handle the animation.
	playDashAnimation()

	rayParams.FilterDescendantsInstances = {character}

	local direction = root.CFrame.LookVector
	local remainingDistance = dashDistance

	local maxBounces = 3
	local bounceCount = 0

	while remainingDistance > 0
		and dashing
		and character.Parent
		and humanoid.Health > 0 do

		local startPosition = root.Position

		local deltaTime = RunService.Heartbeat:Wait()

		local frameDistance = math.min(
			DASH_SPEED * deltaTime,
			remainingDistance
		)

		local result = workspace:Raycast(
			startPosition,
			direction * frameDistance,
			rayParams
		)

		if result then
			local hitPosition = result.Position

			local traveled =
				(hitPosition - startPosition).Magnitude

			local newPosition =
				hitPosition - direction * 0.15

			root.CFrame = CFrame.lookAt(
				newPosition,
				newPosition + direction
			)

			remainingDistance -= traveled

			local newDirection = getBounceDirection(
				direction,
				result.Normal
			)

			if newDirection and bounceCount < maxBounces then
				bounceCount += 1
				direction = newDirection
			else
				break
			end
		else
			local newPosition =
				startPosition + direction * frameDistance

			root.CFrame = CFrame.lookAt(
				newPosition,
				newPosition + direction
			)

			remainingDistance -= frameDistance
		end
	end

	dashing = false

	stopDashAnimation()

	task.wait(COOLDOWN)

	canDash = true
end

dashButton.MouseButton1Click:Connect(dash)
